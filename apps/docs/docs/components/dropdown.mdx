export const closeOnSelectDemo = String.raw`
function CloseOnSelectDemo() {
  const [selected, setSelected] = React.useState(['seat']);

  const toggle = (value) => {
    setSelected((current) =>
      current.includes(value)
        ? current.filter((item) => item !== value)
        : [...current, value],
    );
  };

  return (
    <Dropdown closeOnSelect={false} contentWidth="trigger">
      <Dropdown.Trigger as={Button}>Add-ons ({selected.length})</Dropdown.Trigger>
      <Dropdown.Content>
        {[
          ['seat', 'Extra legroom seat'],
          ['meal', 'Special meal'],
          ['bag', 'Extra baggage'],
        ].map(([value, label]) => (
          <Dropdown.Item key={value} textValue={label} onSelect={() => toggle(value)}>
            <span>{label}</span>
            <span className="inline-flex w-4 justify-center">
              {selected.includes(value) ? <CheckIconOutlinedRounded aria-hidden="true" width={16} height={16} /> : null}
            </span>
          </Dropdown.Item>
        ))}
      </Dropdown.Content>
    </Dropdown>
  );
}

render(<CloseOnSelectDemo />);
`;

# Dropdown

`Dropdown` is an action menu for commands, secondary navigation, and compact
grouped choices that should not write a selected value back into the trigger. It
wraps Spar's headless `DropdownMenu` primitive and adds Takeoff slot classes,
size styling, and compound anatomy.

## Usage

```tsx
import { Dropdown } from '@takeoff-ui/react-spar';
```

```tsx
<Dropdown>
  <Dropdown.Trigger />
  <Dropdown.Content>
    <Dropdown.Item />
    <Dropdown.Separator />
    <Dropdown.Group>
      <Dropdown.Label />
      <Dropdown.Item />
    </Dropdown.Group>
    <Dropdown.Arrow />
  </Dropdown.Content>
</Dropdown>
```

## Accessibility & Keyboard

- Trigger uses `aria-expanded`, `aria-haspopup="menu"`, and `aria-controls`.
- Content renders as a menu and items render as menu items.
- Disabled items remain visible but cannot be highlighted or selected.
- Escape closes the menu and returns focus to the trigger.

| Key                                 | Behavior                                        |
| ----------------------------------- | ----------------------------------------------- |
| <kbd>Enter</kbd> / <kbd>Space</kbd> | Open the menu or activate the highlighted item. |
| <kbd>↓</kbd> / <kbd>↑</kbd>         | Move highlight to the next or previous item.    |
| <kbd>Home</kbd> / <kbd>End</kbd>    | Move highlight to the first or last item.       |
| <kbd>Esc</kbd>                      | Close the menu without selecting an item.       |
